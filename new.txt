#generators > A generator is a special Python function that uses yield to produce values one at a time instead of returning them all at once.
'''Why Use Generators?
Normal functions:return all values at once
Generators:generate values only when needed
This is called:lazy evaluation'''
def numbers():
    yield 1
    yield 2
    yield 3
x = numbers()
print(x)
#getting values
def numbers():
    yield 1
    yield 2
    yield 3
x = numbers()
print(next(x))
print(next(x))
print(next(x))


#function catching > Function caching is a technique where previously computed function results are stored and reused to improve performance. In Python, it is commonly implemented using functools.lru_cache.
#Using lru_cache
from functools import lru_cache
@lru_cache(maxsize=None)
def square(n):
    print("Calculating...")
    return n * n
print(square(4))
print(square(4))
#Second time:result comes from cache,function does not run again
#using cache fibonacci becomes fast
from functools import lru_cache
@lru_cache(maxsize=None)
def fib(n):
    if n <= 1:
        return n
    return fib(n-1) + fib(n-2)
print(fib(50))
 #maxsize means store only 100 recent results
 #maxsize=None > Unlimited cache size.
